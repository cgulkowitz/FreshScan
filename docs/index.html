${escapeHtml(catLabel(rule.cat))}
${escapeHtml(rule.store)} ${escapeHtml(humanDate(expires))} ${flag.label} `; tbody.appendChild(tr); }); tableWrap.classList.remove('hidden'); emptyState.classList.add('hidden'); dateBadge.textContent = "Purchase: " + new Date(purchaseDate.value).toLocaleDateString(undefined,{weekday:'long', month:'short', day:'numeric', year:'numeric'}); /* --- NAME EDIT HANDLER → auto update category, instructions, expiration, colors --- */ $$('#resultTable .nameInput').forEach((inp)=>{ inp.addEventListener('change', ()=>{ const tr = inp.closest('tr'); const newName = inp.value.trim(); const cat = categorize(newName); // update storage instructions const rule = shelfLifeFor(newName, cat); tr.querySelector('.storeCell').textContent = rule.store; // get shelf life const lifeInput = tr.querySelector('.lifeInput'); let nDays = parseInt(lifeInput.value) || rule.days || 7; const base = new Date(purchaseDate.value); const newExp = addDays(base, nDays); // update expiration tr.querySelector('.expiresCell').innerHTML = `${escapeHtml(humanDate(newExp))}`; // update urgency flag const flag = colorFlag(nDays); tr.querySelector('.flagCell').innerHTML = `${flag.label}`; // update category tag tr.querySelector('.tag').textContent = catLabel(cat); }); }); /* existing shelf-life editor listener stays the same */ $$('#resultTable .lifeInput').forEach((inp)=>{ inp.addEventListener('change', ()=>{ const val = inp.value.trim().toLowerCase(); let nDays = 7; const m = val.match(/(\d+(?:\.\d+)?)\s*(day|days|d|month|months|mo)/); if(m){ const num = parseFloat(m[1]); const unit = m[2]; nDays = /mo|month/.test(unit) ? Math.round(num*30) : Math.round(num); } const base = new Date(purchaseDate.value); const newExp = addDays(base, nDays); const row = inp.closest('tr'); row.querySelector('.expiresCell').innerHTML = `${escapeHtml(humanDate(newExp))}`; const flag = colorFlag(nDays); row.querySelector('.flagCell').innerHTML = `${flag.label}`; }); }); }
